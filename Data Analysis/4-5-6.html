import matplotlib.pyplot as plt
import numpy as np

x = np.linspace(0, 10, 100)
y1 = np.sin(x)
y2 = np.cos(x)

# إضافة أكثر من plot لنفس الرسم البياني مع عنوان ولون لكل مجموعة
plt.plot(x, y1, label='sin(x)', color='blue')
plt.plot(x, y2, label='cos(x)', color='orange')

# إظهار مفتاح الرسم وعناوين المحاور والعنوان الرئيسي
plt.legend()
plt.xlabel('X Values')
plt.ylabel('Amplitude')
plt.title('Comparing sin(x) and cos(x) with Matplotlib')
plt.show()

المخطط الشريطي (Bar Plot) والمدرج التكراري (Histogram)

1. المقارنة النظرية وشرح المعاملات

المخطط الشريطي (plt.bar)

يُستخدم لمقارنة قيم المجموعات المختلفة (أفقياً أو رأسياً)، ويلائم البيانات النوعية (مثل تقييمات الأفلام Bad و Good) أو الكمية ذات القيم المحدودة القليلة.

  • x: الفئات المعروضة على المحور الأفقي مثل movie_ratings.
  • height: القيم العددية المقابلة لكل فئة مثل rating_counts.
  • color: لون الأعمدة (مثل 'blue').

المدرج التكراري (plt.hist)

يمثل توزيع متغير رقمي عن طريق تقسيم القيم إلى فترات وحساب التكرارات داخل كل فترة (مثل فترات 20~30، أو توزيع درجات 100 طالب من 0 إلى 100).

  • x: مصفوفة القيم الرقمية المستمرة.
  • bins: عدد الفترات (أو قائمة حدود الفترات) التي تُجمع فيها التكرارات، مثل bins=10.
  • edgecolor: لون حدود الأعمدة لتوضيح الفواصل بين الفترات.

المخطط الدائري والخطي ومخطط الصندوق

1. المفاهيم والمعاملات الأساسية

المخطط الدائري (plt.pie)

يعرض نسب الأجزاء إلى الكل (تُحسب نسبة الجزء بقسمة قيمته على المجموع، مثل 85 / 296 = 28.7% لمادة الرياضيات).

  • colors: قائمة ألوان بعدد الشرائح.
  • startangle: زاوية بداية أول شريحة (مثل 90).
  • autopct: تنسيق النسبة المئوية مثل '%0.1f%%'.
  • explode: قائمة إزاحة لإبراز شريحة معينة عن المركز.
  • shadow: إضافة ظل ثلاثي الأبعاد (True/False).

المخطط الخطي (plt.plot - Line Plot)

يُستخدم لعرض تغير البيانات بمرور الوقت (أو عبر متغير بفترات ثابتة)، ويتكون من خط يربط نقاط البيانات على المحورين x و y لإبراز الاتجاه الزمني.

مخطط الصندوق (plt.boxplot)

يلخص توزيع البيانات الرقمية: الخط الأوسط يمثل الوسيط (Median)، وحدود الصندوق تمثل الربعين الأول والثالث، والدوائر تمثل القيم المتطرفة (Outliers).

  • vert: أفقي عند False ورأسي عند True.
  • widths: عرض الصندوق كنسبة من الشكل.
  • meanline: عرض خط المتوسط الحسابي.
  • showfliers: إظهار أو إخفاء القيم المتطرفة.
  • patch_artist: تفعيل تعبئة الصندوق بلون خلفية.

المخطط النقطي (Scatter Plot) والمقارنة مع Seaborn

1. تمثيل العلاقة في بيانات البطريق (Penguins Dataset)

المخطط النقطي في Matplotlib

يمثل العلاقة بين متغيرين رقميين مثل culmen_length_mm و body_mass_g عبر نقاط على الإحداثيين x و y (تظهر علاقة طردية واضحة).

  • s: التحكم بحجم النقاط (مثل متغير psize: 10 للذكور و 100 للإناث).
  • c / color: تخصيص اللون عبر قاموس colors وحلقة تكرار لكل فئة.
  • marker: شكل النقطة (مثل 'o' و '^') بناءً على عمود sex.

الاختصار الاحترافي عبر Seaborn

تغني دالة sns.scatterplot() عن كتابة حلقات التكرار اليدوية والقاموس، إذ تتيح الترميز اللوني والشكلي مباشرة عبر اسم العمود:

  • hue='sex': تلوين النقاط تلقائياً حسب الفئة.
  • palette: تحديد لوحة الألوان (مثل 'Set1' أو قاموس ألوان).
  • style='sex': تغيير شكل النقطة تلقائياً حسب الفئة.
  • markers: تحديد قاموس الأشكال مثل {'MALE':'s', 'FEMALE':'o'}.

2. مقارنة الكود: Matplotlib اليدوي مقابل Seaborn المختصر

Python · plt.scatter() + Loop
import matplotlib.pyplot as plt
import pandas as pd

df = pd.read_csv('penguins.csv').dropna()
colors = {'MALE': 'blue', 'FEMALE': 'red'}
markers = {'MALE': 's', 'FEMALE': 'o'}
sizes = {'MALE': 10, 'FEMALE': 100}

for sex_val in ['MALE', 'FEMALE']:
    sub = df[df['sex'] == sex_val]
    plt.scatter(sub['culmen_length_mm'], sub['body_mass_g'],
                s=sizes[sex_val], c=colors[sex_val],
                marker=markers[sex_val], label=sex_val)

plt.xlabel('culmen_length_mm')
plt.ylabel('body_mass_g')
plt.title('Penguins Scatter (Matplotlib)')
plt.legend()
plt.show()

جداول التقاطع والمحورية، ورسوم المتغيرين، والارتباط الحراري

1. الجداول الإحصائية والرسوم الثنائية ومعامل الارتباط

جدول التقاطع (pd.crosstab)

يعرض العلاقة بين متغيرين نوعيين على شكل مصفوفة تكرارات (مثل أعداد الذكور والإناث في كل فصيلة بطريق: Adelie 73/73، Chinstrap 34/34، Gentoo 58/62).

الجدول المحوري (pd.pivot_table)

يمثل متغيرين أو أكثر (أحدهما نوعي في index والباقي كمي في values) مع دوال تجميع في aggfunc مثل ['mean', 'sum'].

رسوم متغيرين في Seaborn

  • sns.barplot: المحور x فئات نوعية (species) والمحور y قيمة إحصائية (مثل الوسيط لـ culmen_length_mm).
  • sns.boxplot: المحور x فئات نوعية والمحور y توزيع كل فئة.

الارتباط (Correlation) والخريطة الحرارية

يتراوح معامل الارتباط ضمن [-1, +1]: قيمة +1 ارتباط طردي مثالي، -1 عكسي مثالي، 0 لا ارتباط (مثلاً 0.589 علاقة طردية قوية). يُحسب بـ .corr() أو .corr(numeric_only=True) ويُرسم بـ sns.heatmap.

2. أكواد بايثون كاملة (Crosstab, Pivot, Correlation, Heatmap)

Python · crosstab, pivot_table & Bivariate Plots
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# 1) جدول التقاطع لمتغيرين نوعيين
ct = pd.crosstab(df['species'], df['sex'])

# 2) الجدول المحوري
pt = pd.pivot_table(
    df,
    index='species',
    values=['culmen_length_mm', 'body_mass_g'],
    aggfunc=['mean', 'sum']
)

# 3) barplot و boxplot لمتغيرين
sns.barplot(data=df, x='species', y='culmen_length_mm', estimator='median')
plt.show()
sns.boxplot(data=df, x='species', y='culmen_length_mm')
plt.show()
Python · .corr() & sns.heatmap()
import seaborn as sns
import matplotlib.pyplot as plt

تخطيطات العرض في Bokeh: row() و column() و gridplot()

1. دوال التخطيط وترتيب الرسوم المتعددة

row(p1, p2, ...)

ترتب عدة رسوم بيانية جنباً إلى جنب في صف أفقي واحد، وهي مثالية للمقارنة السريعة بين مخططين لهما نفس المقياس الرأسي.

column(p1, p2, ...)

ترتب الرسوم البيانية فوق بعضها في عمود رأسي واحد، وتناسب السلاسل الزمنية التي تشترك في نفس المحور الأفقي x.

gridplot([[p1, p2], [p3, None]])

شريط الأدوات والتلميحات التفاعلية ومفتاح الخريطة

1. التحكم بالتفاعل في Bokeh

موضع وتخصيص شريط الأدوات

  • toolbar_location: يقبل 'above', 'below', 'left', 'right' أو None لإخفائه.
  • tools و add_tools(): لتحديد أو إضافة أدوات مثل FullscreenTool و BoxZoomTool و ResetTool.

التلميحات التفاعلية (HoverTool / Tooltips)

تعرض بيانات النقطة عند تمرير المؤشر عبر قائمة tooltips:

  • @field: يعرض قيمة العمود من المصدر.
  • $color[hex]:fill_color: يعرض اللون بالنظام السداسي عشري.
  • $color[swatch]:fill_color: يعرض مربعاً ملوناً باللون الفعلي للنقطة.

مفتاح الخريطة التفاعلي (Legend Click)

يتيح إخفاء السلاسل أو تخفيتها عند النقر على اسمها في الـ legend عبر تعيين p.legend.click_policy = 'hide' أو 'mute'.

2. كود بايثون كامل (Toolbar, HoverTool & Click Policy)

Python · HoverTool, add_tools & click_policy
from bokeh.plotting import figure, show
from bokeh.models import HoverTool, FullscreenTool, BoxZoomTool, ResetTool

# 1) تحديد موضع شريط الأدوات والأدوات المبدئية
p = figure(title='Interactive Bokeh Plot',
           toolbar_location='above',
           tools='pan,wheel_zoom')

# 2) إضافة أدوات مخصصة عبر add_tools
p.add_tools(BoxZoomTool(), ResetTool(), FullscreenTool())

# 3) إضافة التلميحات التفاعلية HoverTool
hover = HoverTool(tooltips=[
    ('Species', '@species'),
    ('Sepal', '@sepal_length'),
    ('Color Hex', '$color[hex]:fill_color'),
    ('Swatch', '$color[swatch]:fill_color')
])
p.add_tools(hover)

# 4) تفعيل إخفاء أو تخفيت الخطوط بالنقر على الـ legend
p.line([1, 2, 3, 4], [10, 14, 12, 18], legend_label='Series A', line_width=3, color='blue')
p.line([1, 2, 3, 4], [15, 11, 16, 13], legend_label='Series B', line_width=3, color='orange')
p.legend.click_policy = 'hide'  # أو 'mute'
show(p)

3. مختبر تفاعلي حي: جرب شريط الأدوات والتلميحات والنقر على Legend

مرر المؤشر أو انقر على إحدى النقاط الدائرية لمعاينة HoverTool، أو انقر على Series A/B في الـ Legend.
